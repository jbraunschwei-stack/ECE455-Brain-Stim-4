cmake_minimum_required(VERSION 3.13)
project(dds_host_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
add_executable(test_dds test_dds.c ../dds_control.c ../dds_tables.c)
target_include_directories(test_dds PRIVATE ..)
add_executable(test_spi test_spi.c ../spi_dac.c)
target_include_directories(test_spi PRIVATE mocks ..)
if(NOT MSVC)
    target_compile_options(test_dds PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_compile_options(test_spi PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(test_dds m)
endif()
add_test(NAME dds_math_and_protocol COMMAND test_dds)
add_test(NAME spi_framing COMMAND test_spi)
